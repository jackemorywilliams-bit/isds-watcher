```python
import json, re, collections
rows = [json.loads(l) for l in open('analytics/insights.jsonl') if l.strip()]
def klass(r):
    v = str(r.get('ring', '')).strip()
    if not v or v == '?': return 'UNLABELLED'
    lv = v.lower()
    inst = bool(re.search(r'instrument|method|architecture|record/doctrine|process'
                          r'|digest semantics|source health', lv))
    ring = bool(re.search(r'ring ?[123]|^[123]$|ring1|ring2|ring3', lv))
    if ring and inst: return 'BOTH'
    if ring: return 'RESEARCH QUESTION'
    if inst: return 'INSTRUMENT/METHOD'
    return 'OTHER(' + v + ')'
print(collections.Counter(klass(r) for r in rows).most_common())
```
